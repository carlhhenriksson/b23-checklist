// Bristell B23 checklist - offline service worker (cache-first for a self-contained app).
// Note: service workers require https or localhost; they do not run from file://.
const CACHE = 'b23-checklist-v13';
const ASSETS = ['./', './index.html', './manifest.webmanifest', './icon.svg'];

self.addEventListener('install', function (event) {
  event.waitUntil(
    caches.open(CACHE).then(function (cache) {
      // Cache each asset independently so one missing file does not abort the whole install.
      return Promise.all(ASSETS.map(function (asset) { return cache.add(asset).catch(function () {}); }));
    }).then(function () { return self.skipWaiting(); })
  );
});

self.addEventListener('activate', function (event) {
  event.waitUntil(
    caches.keys().then(function (keys) {
      return Promise.all(keys.map(function (key) { if (key !== CACHE) return caches.delete(key); }));
    }).then(function () { return self.clients.claim(); })
  );
});

self.addEventListener('fetch', function (event) {
  if (event.request.method !== 'GET') return;
  event.respondWith(
    caches.match(event.request).then(function (cached) {
      return cached || fetch(event.request).then(function (response) {
        const copy = response.clone();
        caches.open(CACHE).then(function (cache) { cache.put(event.request, copy).catch(function () {}); });
        return response;
      }).catch(function () { return cached; });
    })
  );
});
